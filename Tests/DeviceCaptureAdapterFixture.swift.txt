@MainActor enum TargetActivator {
    enum Outcome { case alreadyFront, activated, launched, failed }
    static var front: TargetKind? = .chatGPT
    static var alwaysReady = true
    static var activations = 0
    static var delay: UInt64 = 0
    static func observe(_ kind: TargetKind) -> Bool { alwaysReady || front == kind }
    static func bringForward(_ kind: TargetKind, shouldContinue: () -> Bool) async throws -> Outcome {
        if observe(kind) { return .alreadyFront }
        activations += 1
        if delay > 0 { try await Task.sleep(nanoseconds: delay) }
        guard shouldContinue() else { return .failed }
        front = kind; return .activated
    }
}

@MainActor final class FakeTarget {
    var lease = Lease(name: "initial")
    var unavailable = false
    var requiresFocus = false
    var focusRecoveries = 0
    var focusDelay: UInt64 = 0
    func locateLease() throws -> any ChatLease {
        if unavailable || requiresFocus { throw TargetFailure.currentThreadUnavailable }; return lease
    }
    func locateForCaptureLease(shouldContinue: (Int32) -> Bool) async throws -> any ChatLease {
        if requiresFocus {
            guard shouldContinue(1) else { throw TargetFailure.targetChanged }
            focusRecoveries += 1
            if focusDelay > 0 { try await Task.sleep(nanoseconds: focusDelay) }
            guard shouldContinue(1) else { throw TargetFailure.targetChanged }
            requiresFocus = false
        }
        return try locateLease()
    }
    func raiseLeaseWindow(_ lease: any ChatLease) async throws {}
    func openLeaseConversation(_ lease: any ChatLease) throws { self.lease = Lease(name: lease.name) }
}
@MainActor final class FakeWorker {
    struct Config { let deviceClass: DeviceClass }
    struct Pending { let id: String; let file: URL; let done: (Data, Int32) -> Void }
    let kind: DeviceClass
    var pending: Pending?
    var starts: [String] = []
    var onCaptureStart: ((String) -> Void)?
    var isBusy: Bool { pending != nil }
    struct Connection { enum State { case ready, failed }; var state = State.ready }
    var connectionStatus = Connection()
    var deviceLocked: Bool?
    init(_ kind: DeviceClass) { self.kind = kind }
    func capture(configuration: Config, requestID: String, file: URL, completion: @escaping (Data, Int32) -> Void) throws {
        precondition(configuration.deviceClass == kind, "worker device mismatch")
        precondition(pending == nil, "overlapping same-device RPC")
        precondition(file.lastPathComponent.hasPrefix(kind.rawValue + "-"), "wrong image device filename")
        onCaptureStart?(requestID)
        starts.append(requestID); pending = Pending(id: requestID, file: file, done: completion)
    }
    func finish(reason: String? = nil) {
        guard let request = pending else { preconditionFailure("no pending RPC") }
        pending = nil
        let data = try! JSONSerialization.data(withJSONObject: ["schema_version": 1, "request_id": request.id,
            "ok": reason == nil, "reason": reason ?? "captured", "device_class": kind.rawValue])
        request.done(data, reason == nil ? 0 : 1)
    }
}
